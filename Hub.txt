local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local UserInputService = game:GetService("UserInputService")

local GUI_NAME = "Hub4NewScriptMenu"

local NEW_SCRIPT = [[script_key = "LP-0F96759D9B66FB7E";
loadstring(game:HttpGet("https://cdn.luaprotect.dev/u/3e0424/Fp-EsEyBmU0lo1l9"))()]]

local TR_TEXT = "Bu script değişti, yeni script şu:"

local CoreGui
pcall(function()
    CoreGui = game:GetService("CoreGui")
end)

local playerGui = LocalPlayer:WaitForChild("PlayerGui")

local guiParent
pcall(function()
    if gethui then
        guiParent = gethui()
    end
end)
if not guiParent then
    guiParent = CoreGui
end
if not guiParent then
    guiParent = playerGui
end

-- Eski menü açıksa kaldır
local parents = { playerGui }
if CoreGui then
    table.insert(parents, 1, CoreGui)
end
if guiParent ~= CoreGui and guiParent ~= playerGui then
    table.insert(parents, 1, guiParent)
end
for _, parent in ipairs(parents) do
    local old = parent:FindFirstChild(GUI_NAME)
    if old then
        old:Destroy()
    end
end

-- ---------------------------------------------------------------
-- Yardımcılar
-- ---------------------------------------------------------------
local function addCorner(instance, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius)
    c.Parent = instance
    return c
end

local function addStroke(instance, color, thickness)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness or 1
    s.Parent = instance
    return s
end

local function copyToClipboard(text)
    local fn
    pcall(function()
        fn = setclipboard or toclipboard
    end)
    if type(fn) == "function" then
        local ok = pcall(fn, text)
        if ok then
            return true
        end
    end
    return false
end

-- Sürükleme
local dragging = false
local dragStart = nil
local startPos = nil

header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
    end
end)

header.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
