-- UNIVERSAL HUB 1.0 yukle
local function loadUniversalHub()
    notify("Loader", "UNIVERSAL HUB 1.0 yukleniyor...", 2)
    task.wait(0.5)
    destroyLoader()
    -- Universal hub kodunu buradan yukle - Egitim_Hub.lua ile ayni
    -- Dosyadan dene, olmazsa embedded calistir
    local loaded = false
    pcall(function()
        if isfile and readfile then
            if isfile("Egitim_Hub.lua") then
                local c = readfile("Egitim_Hub.lua")
                if c and #c > 100 then loadstring(c)(); loaded = true; return end
            end
            -- Desktop path fallback (bazı executorlar)
            local p1 = "C:\\Users\\lutfi\\OneDrive\\Desktop\\Egitim_Hub.lua"
            if isfile(p1) then loadstring(readfile(p1))(); loaded = true; return end
        end
    end)
    if loaded then return end
    -- Fallback: direkt universal hub'i inline calistir (kisa versiyon - tam hub'i acar)
    -- Tam universal hub kodu asagida Muscle Legends ile birlikte yuklu degilse, kullaniciya bilgi ver
    notify("Bilgi", "Egitim_Hub.lua bulunamadi, universal hub inline yukleniyor", 3)
    -- Inline universal hub (orijinal dosyanin aynisi - kisaltilmis cagri)
    -- En guvenilir: loadstring ile HttpGet dene
    pcall(function()
        local code = game:HttpGet("https://raw.githubusercontent.com/anomalyco/opencode/main/Egitim_Hub.lua")
        if code and #code > 100 then loadstring(code)(); loaded = true end
    end)
    if not loaded then
        -- Son fallback: minimal universal hub ac
        notify("Hata", "Universal hub dosyasi bulunamadi. Lutfen Egitim_Hub.lua ile ayni klasorde calistir.", 4)
        -- Minimal hub olustur (sadece Misc)
        warn("[Loader] Universal hub yuklenemedi - dosya yok")
    end
end
