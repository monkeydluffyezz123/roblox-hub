local function disconnectAll(list)
    if not list then
        return
    end
    for _, conn in ipairs(list) do
        pcall(function()
            conn:Disconnect()
        end)
    end
end

-- =========================================================
-- HELPERS

local function new(className, props, parent)
    local inst = Instance.new(className)
    if props then
        for key, value in pairs(props) do
            inst[key] = value
        end
    end
    if parent then
        inst.Parent = parent
    end
    return inst
end

local function round(parent, radius)
    return new("UICorner", { CornerRadius = UDim.new(0, radius or 8) }, parent)
end

local function stroke(parent, color, thickness)
    return new("UIStroke", {
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Color = color,
        Thickness = thickness or 1,
    }, parent)
end

local function grad(parent, color1, color2, rotation)
    return new("UIGradient", {
        Color = ColorSequence.new(color1, color2),
        Rotation = rotation or 0,
    }, parent)
end

local function tween(inst, duration, props, easing)
    local ok, animation = pcall(function()
        return TweenService:Create(
            inst,
            TweenInfo.new(duration or 0.18, easing or Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            props
        )
    end)
    if ok and animation then
        pcall(function()
            animation:Play()
        end)
        return animation
    end
    return nil
end

local function copyText(text)
    if setclipboard then
        local ok = pcall(setclipboard, text)
        if ok then
            return true
        end
    end
    return false
end

local function makeDiscordButton(parent, y)
    local btn = new("TextButton", {
        AutoButtonColor = false,
        BackgroundColor3 = Color3.fromRGB(88, 101, 242),
        BorderSizePixel = 0,
        Font = Enum.Font.GothamBold,
        Position = UDim2.new(1, -92, 0, y or 10),
        Size = UDim2.new(0, 56, 0, 20),
        Text = LHt("discord"),
        TextColor3 = Color3.fromRGB(255, 255, 255),
        TextSize = 9,
    }, parent)
    round(btn, 5)
    bind(btn.Activated:Connect(function()
        local copied = copyText(DISCORD_LINK)
        btn.Text = copied and LHt("copied") or LHt("copy_failed")
        task.delay(1.2, function()
            if btn.Parent then
                btn.Text = LHt("discord")
            end
        end)
    end))
    return btn
end

local function makeDraggable(frame, handle)
    local dragging = false
    local dragStart = nil
    local startPos = nil

    bind(handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end))

    bind(handle.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end))

    bind(UserInputService.InputChanged:Connect(function(input)
        if not frame.Parent then
            dragging = false
            return
        end
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end))
end

local LH_Registry = {}

local function LHt(key)
    local t = LH_TEXT[LH_LANG] or LH_TEXT.EN
    return t[key]
end

local function LHLines()
    local t = LH_TEXT[LH_LANG] or LH_TEXT.EN
    return t.update_lines
end

local function LHRegister(label, key)
    if not label then return end
    LH_Registry[#LH_Registry + 1] = { label = label, key = key }
end

local function LHApplyLanguage()
    for _, item in ipairs(LH_Registry) do
        if item.label and item.label.Parent then
            local value = LHt(item.key)
            if value ~= nil then
                item.label.Text = value
            end
        end
    end
end

local function LHSetLanguage(lang)
    if lang ~= "EN" and lang ~= "TR" then return end
    LH_LANG = lang
    pcall(function() _G.LIFTHUB_LANG = lang end)
    pcall(function() writefile("lifthub_lang.txt", lang) end)
    LHApplyLanguage()
end

local function LHToggleLanguage()
    LHSetLanguage(LH_LANG == "EN" and "TR" or "EN")
end

        local lines = LHLines()
        scroll.CanvasSize = UDim2.new(0, 0, 0, math.max(300, #lines * 22 + 16))
        for _, label in ipairs(lineLabels) do
            if label.Parent then label:Destroy() end
        end
        lineLabels = {}
        for i, line in ipairs(lines) do
            local lineColor = Color3.fromRGB(200, 210, 218)
            if i == 1 or string.find(line, "NEW", 1, true) or string.find(line, "YENİ", 1, true) then
                lineColor = Color3.fromRGB(0, 214, 152)
            elseif string.find(line, "INFO", 1, true) or string.find(line, "BİLGİ", 1, true) then
                lineColor = Color3.fromRGB(0, 168, 232)
            elseif string.find(line, "ARAYÜZ", 1, true) or string.find(line, "UI", 1, true) then
                lineColor = Color3.fromRGB(255, 190, 60)
            elseif string.find(line, "---", 1, true) then
                lineColor = Color3.fromRGB(52, 68, 78)
            end
            local label = new("TextLabel", {
                BackgroundTransparency = 1,
                Font = Enum.Font.Gotham,
                Position = UDim2.new(0, 4, 0, (i - 1) * 22 + 4),
                Size = UDim2.new(1, -8, 0, 18),
                Text = line,
                TextColor3 = lineColor,
                TextSize = 9,
                TextXAlignment = Enum.TextXAlignment.Left,
                ZIndex = z + 5,
            }, scroll)
            lineLabels[i] = label
        end
        settingsButton.Text = LHt("nav_settings") .. " (" .. LHt("lang_button") .. ")"
        if opts.onLanguageChange then opts.onLanguageChange() end
    end
    refreshPanel()

    local function hoverFooter(button, highlight)
        local idle = Color3.fromRGB(23, 31, 38)
        local enter = function()
            tween(button, 0.12, { BackgroundColor3 = highlight })
        end
        local leave = function()
            tween(button, 0.12, { BackgroundColor3 = idle })
        end
        if bind then
            bind(button.MouseEnter:Connect(enter))
            bind(button.MouseLeave:Connect(leave))
        else
            button.MouseEnter:Connect(enter)
            button.MouseLeave:Connect(leave)
        end
    end
    hoverFooter(updatesButton, Color3.fromRGB(30, 52, 60))
    hoverFooter(settingsButton, Color3.fromRGB(26, 52, 45))

    local function wire(button, fn)
        if bind then
            bind(button.Activated:Connect(fn))
        else
            button.Activated:Connect(fn)
        end
    end

    wire(updatesButton, function()
        panel.Visible = not panel.Visible
        if panel.Visible then
            refreshPanel()
            panelScale.Scale = 0.9
            tween(panelScale, 0.22, { Scale = 1 }, Enum.EasingStyle.Back)
        end
    end)
    wire(panelExtra, function()
        local hook = _G.LIFTHUB_OPEN_XTRA
        if type(hook) == "function" then
            hook()
        else
            pcall(function()
                if setclipboard then setclipboard(AD_LINK) end
            end)
        end
    end)
    wire(settingsButton, function()
        LHToggleLanguage()
        refreshPanel()
    end)
    wire(panelClose, function()
        panel.Visible = false
    end)

    return panel
end

    local function watchContainer(container)
        if not container then
            return

        for _, known in ipairs(watchedContainers) do
            if known == container then
                return

        end)
    end

    local function readScript(entry)
        if not readfile then
            return nil, LHt("err_readfile")
        end
        local candidates = {
            absolutePath(entry),
            entry.file,
            "scripts/" .. entry.file,
            "autoexec/" .. entry.file,
        }
        for _, path in ipairs(candidates) do
            local ok, source = pcall(readfile, path)
            if ok and type(source) == "string" and #source > 10 then
                return source, path
            end
        end
        return nil, LHt("err_not_found") .. entry.file
    end

    local function openScript(entry)
        local source, pathOrError = readScript(entry)
        if not source then
            setStatus(tostring(pathOrError) .. LHt("err_copy_hint"), RED)
            return
        end

        local loader = loadstring or load
        if not loader then
            setStatus(LHt("err_loadstring"), RED)
            return
        end

        closeWatchUntil = os.clock() + 8
        local chunk, compileError = loader(source, entry.file)
        if not chunk then
            setStatus(LHt("err_compile") .. tostring(compileError), RED)
            return
        end

        setStatus(LHt("status_opening") .. entry.name, BLUE)
        task.spawn(function()
            local ok, runError = pcall(chunk)
            if not ok then
                setStatus(LHt("status_error") .. tostring(runError), RED)
            else
                setStatus(LHt("status_opened") .. entry.name, GREEN)
            end
        end)
    end
